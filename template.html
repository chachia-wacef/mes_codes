import boto3 import json # 1. Initialiser le client Bedrock Runtime # (Utilisez la région où vos modèles sont activés) bedrock_client = boto3.client("bedrock-runtime", region_name="us-east-1") # Définir le modèle exact (ex: Claude 3.5 Haiku) # Note : L'API CountTokens requiert l'ID du modèle de base (pas un ARN de profil de région) model_id = "anthropic.claude-3-5-haiku-20241022-v1:0" texte_X = "Voici votre texte ou votre morceau de transcription de 50k..." # ===================================================================== # OPTION A : Si vous utilisez la méthode globale .converse() [Recommandé] # ===================================================================== input_converse = { "messages": [ { "role": "user", "content": [{"text": texte_X}] } ] } response_converse = bedrock_client.count_tokens( modelId=model_id, input={"converse": input_converse} ) print(f"Tokens via Converse API : {response_converse['inputTokens']}") # ===================================================================== # OPTION B : Si vous utilisez la méthode classique .invoke_model() # ===================================================================== # Le format du body dépend du modèle (ici le format pour Anthropic Claude) body_structure = json.dumps({ "anthropic_version": "bedrock-2023-05-31", "max_tokens": 1000, "messages": [{"role": "user", "content": texte_X}] }) response_invoke = bedrock_client.count_tokens( modelId=model_id, input={"invokeModel": {"body": body_structure}} ) print(f"Tokens via InvokeModel API : {response_invoke['inputTokens']}")
